services:
  ecommerce_Db:
    image: postgres:16-alpine
    container_name: ecommerce_db
    restart: unless-stopped
    environment:
      POSTGRES_DB: Ecommerce
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: 12345678
    ports:
      - "5433:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U postgres -d Ecommerce"]
      interval: 10s
      timeout: 5s
      retries: 5

  rabbitmq:
    image: rabbitmq:3.13-management-alpine
    container_name: ecommerce_rabbitmq
    restart: unless-stopped
    ports:
      - "5672:5672"
      - "15672:15672"
    environment:
      RABBITMQ_DEFAULT_USER: guest
      RABBITMQ_DEFAULT_PASS: guest
    volumes:
      - rabbitmq_data:/var/lib/rabbitmq
    healthcheck:
      test: ["CMD", "rabbitmq-diagnostics", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5

  webapi:
    image: samuelshonekan/ecommerce_webapi:latest
    container_name: ecommerce_api
    restart: unless-stopped
    depends_on:
      ecommerce_Db:
        condition: service_healthy
      rabbitmq:
        condition: service_healthy
    ports:
      - "8080:8080"
    environment:
      ASPNETCORE_ENVIRONMENT: Development
      ASPNETCORE_URLS: http://+:8080
      ConnectionStrings__DefaultConnection: "Host=ecommerce_Db;Port=5432;Database=Ecommerce;Username=postgres;Password=12345678;"
      JwtSettings__SecretKey: "secretkeyYOURSECRETKEYTHATISATLEAST32CHRACTERSLOG"
      JwtSettings__Issuer: "EcommerceWebAPI"
      JwtSettings__Audience: "EcommerceWebAPI"
      JwtSettings__ExpirationMinutes: "60"
      SeedDatabase: "true"
      RabbitMQ__Host: rabbitmq
      RabbitMQ__Username: guest
      RabbitMQ__Password: guest

volumes:
  postgres_data:
  rabbitmq_data:
